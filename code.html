How to get marks on a code question
  1. Start with imports and the function/class header. That alone often earns marks.
  2. Write shapes as comments, e.g. # (B, 32, 16, 16). Examiners love this, and it stops shape bugs.
  3. Follow the same order every time: forward → loss → backward → update (or zero_grad → loss → backward → step).
  4. A tiny syntax slip is fine. A correct idea and the right steps matter most.

1. Decision Tree

DT-1Write gini(y) and entropy(y) in NumPy.
Answer
import numpy as np

def gini(y):
    _, counts = np.unique(y, return_counts=True)
    p = counts / len(y)                  # class proportions
    return 1 - np.sum(p ** 2)

def entropy(y):
    _, counts = np.unique(y, return_counts=True)
    p = counts / len(y)
    return -np.sum(p * np.log2(p))       # every count > 0, so no log(0)

Memory hook: both start with the same 2 lines (counts → proportions p). Only the last line differs.

DT-2Write best_split(X, y) that tries midpoint thresholds on every feature and returns the best (feature, threshold) by weighted Gini.
Answer
def best_split(X, y):
    n = len(y)
    best_score, best_f, best_t = np.inf, None, None
    for f in range(X.shape[1]):                      # every feature
        values = np.unique(X[:, f])                  # sorted unique values
        for t in (values[:-1] + values[1:]) / 2:     # midpoints
            left = X[:, f] <= t
            right = ~left
            score = (left.sum() / n) * gini(y[left]) + (right.sum() / n) * gini(y[right])
            if score < best_score:
                best_score, best_f, best_t = score, f, t
    return best_f, best_t

Memory hook: 2 loops (feature, threshold) → mask left/right → weighted Gini → keep the smallest.

DT-3Write a Node class, a recursive build_tree with stopping rules, and predict_one.
Answer
class Node:
    def __init__(self, feature=None, threshold=None, left=None, right=None, value=None):
        self.feature, self.threshold = feature, threshold
        self.left, self.right = left, right
        self.value = value                     # set only for leaves

def majority(y):
    vals, counts = np.unique(y, return_counts=True)
    return vals[np.argmax(counts)]

def build_tree(X, y, depth=0, max_depth=5, min_samples=2):
    # stopping rules: pure, too few samples, too deep
    if len(np.unique(y)) == 1 or len(y) < min_samples or depth >= max_depth:
        return Node(value=majority(y))
    f, t = best_split(X, y)
    if f is None:                              # no valid split
        return Node(value=majority(y))
    left = X[:, f] <= t
    return Node(f, t,
                build_tree(X[left], y[left], depth + 1, max_depth, min_samples),
                build_tree(X[~left], y[~left], depth + 1, max_depth, min_samples))

def predict_one(node, x):
    while node.value is None:                  # walk down until a leaf
        node = node.left if x[node.feature] <= node.threshold else node.right
    return node.value

def predict(root, X):
    return np.array([predict_one(root, x) for x in X])

2. Random Forest

RF-1Write a bootstrap sample function and a column-wise majority vote.
Answer
def bootstrap_sample(X, y, rng):
    n = len(y)
    idx = rng.choice(n, size=n, replace=True)      # WITH replacement
    return X[idx], y[idx]

def majority_vote(all_preds):
    # all_preds: (n_trees, n_samples) -> one vote per sample (column)
    return np.array([majority(col) for col in all_preds.T])
RF-2Write a RandomForest class with fit and predict using DecisionTreeScratch (from Lab 7).
Answer
class RandomForest:
    def __init__(self, n_trees=30, max_depth=5, seed=42):
        self.n_trees, self.max_depth = n_trees, max_depth
        self.rng = np.random.default_rng(seed)
        self.trees = []

    def fit(self, X, y):
        self.trees = []
        for _ in range(self.n_trees):
            Xb, yb = bootstrap_sample(X, y, self.rng)          # 1. random rows
            tree = DecisionTreeScratch(max_depth=self.max_depth,
                                       max_features="sqrt",      # 2. random features per split
                                       random_state=int(self.rng.integers(1_000_000)))
            tree.fit(Xb, yb)
            self.trees.append(tree)
        return self

    def predict(self, X):
        all_preds = np.array([t.predict(X) for t in self.trees])  # (n_trees, n_samples)
        return majority_vote(all_preds)                           # 3. vote

Memory hook: loop over trees: bootstrap → tree with max_features="sqrt" → fit. Predict = stack → vote.

3. KNN

KNN-1Write knn_predict(X_train, y_train, X_test, k) from scratch (Euclidean distance, majority vote).
Answer
import numpy as np

def knn_predict(X_train, y_train, X_test, k=3):
    preds = []
    for x in X_test:
        d = np.sqrt(np.sum((X_train - x) ** 2, axis=1))   # distance to every train point
        nearest = np.argsort(d)[:k]                        # indices of k closest
        vals, counts = np.unique(y_train[nearest], return_counts=True)
        preds.append(vals[np.argmax(counts)])              # majority vote
    return np.array(preds)

Memory hook: distance → argsort → first k → vote. 4 lines inside the loop.

Answer
from sklearn.model_selection import cross_val_score

scores = {}
for k in range(1, 32, 2):                  # odd K avoids ties
    model = Pipeline([("scaler", StandardScaler()),
                      ("knn", KNeighborsClassifier(n_neighbors=k))])

best_k = max(scores, key=scores.get)

4. Neural Networks

NN-1NumPy: train a 1-hidden-layer network (ReLU hidden, sigmoid output, BCE loss) with full-batch gradient descent. Write forward, backward and update.
Answer
import numpy as np

def sigmoid(z):
    return 1 / (1 + np.exp(-z))

n, d = X.shape                    # X: (n, d), y: (n,) with 0/1
h, lr = 16, 0.1
rng = np.random.default_rng(0)
W1 = rng.normal(0, np.sqrt(2 / d), (d, h)); b1 = np.zeros(h)   # He init (ReLU)
W2 = rng.normal(0, np.sqrt(1 / h), (h, 1)); b2 = np.zeros(1)

for epoch in range(1000):
    # ---- forward ----
    Z1 = X @ W1 + b1              # (n, h)
    A1 = np.maximum(0, Z1)        # ReLU
    Z2 = A1 @ W2 + b2             # (n, 1)
    A2 = sigmoid(Z2).ravel()      # (n,)  predicted probability
    loss = -np.mean(y * np.log(A2 + 1e-12) + (1 - y) * np.log(1 - A2 + 1e-12))

    # ---- backward ----
    dZ2 = (A2 - y).reshape(-1, 1) / n     # sigmoid + BCE shortcut: a - y
    dW2 = A1.T @ dZ2;  db2 = dZ2.sum(axis=0)
    dA1 = dZ2 @ W2.T                      # (n, h)
    dZ1 = dA1 * (Z1 > 0)                  # ReLU derivative
    dW1 = X.T @ dZ1;   db1 = dZ1.sum(axis=0)

    # ---- update ----
    W1 -= lr * dW1;  b1 -= lr * db1
    W2 -= lr * dW2;  b2 -= lr * db2

pred = (A2 >= 0.5).astype(int)

Memory hook: backward goes in reverse: output error a − y → dW = input.T @ dZ → push back with @ W.T → multiply by activation derivative.

NN-2PyTorch: write an MLP class for binary classification and its training + test-accuracy loop.
Answer
import torch
import torch.nn as nn
from torch.utils.data import TensorDataset, DataLoader

class MLP(nn.Module):
    def __init__(self, n_inputs, hidden=64):
        super().__init__()
        self.net = nn.Sequential(
            nn.Linear(n_inputs, hidden), nn.ReLU(),
            nn.Linear(hidden, hidden), nn.ReLU(),
            nn.Linear(hidden, 1),              # raw logit (no sigmoid)
        )

    def forward(self, x):
        return self.net(x).squeeze(1)          # (B,)

X_t = torch.tensor(X_train, dtype=torch.float32)
y_t = torch.tensor(y_train, dtype=torch.float32)
loader = DataLoader(TensorDataset(X_t, y_t), batch_size=32, shuffle=True)

model = MLP(X_t.shape[1])
loss_fn = nn.BCEWithLogitsLoss()               # sigmoid + BCE together
optimizer = torch.optim.Adam(model.parameters(), lr=1e-3)

for epoch in range(50):
    model.train()
    for xb, yb in loader:
        optimizer.zero_grad()
        loss = loss_fn(model(xb), yb)
        loss.backward()
        optimizer.step()

model.eval()
with torch.no_grad():
    X_te = torch.tensor(X_test, dtype=torch.float32)
    preds = (torch.sigmoid(model(X_te)) >= 0.5).long()
    acc = (preds == torch.tensor(y_test)).float().mean().item()

Multi-class instead? Last layer nn.Linear(hidden, num_classes), no squeeze, nn.CrossEntropyLoss() with integer labels, predict with .argmax(dim=1).

NN-3Write one update step for SGD with Momentum, RMSprop and Adam (one weight array w, gradient g).
Answer
# SGD with momentum
v = beta * v + g
w = w - lr * v

# RMSprop
s = rho * s + (1 - rho) * g ** 2
w = w - lr * g / (np.sqrt(s) + eps)

# Adam (t starts at 1)
m = b1 * m + (1 - b1) * g
s = b2 * s + (1 - b2) * g ** 2
m_hat = m / (1 - b1 ** t)          # bias correction
s_hat = s / (1 - b2 ** t)
w = w - lr * m_hat / (np.sqrt(s_hat) + eps)

Adam = Momentum (m) + RMSprop (s) + bias correction. Defaults: b1=0.9, b2=0.999, eps=1e-8.

5. CNN

CNN-1Write a PyTorch CNN for 32×32 RGB images and 10 classes: two conv blocks (Conv-ReLU-MaxPool) then a classifier. Comment the shapes.
Answer
import torch
import torch.nn as nn

class SimpleCNN(nn.Module):
    def __init__(self, num_classes=10):
        super().__init__()
        self.features = nn.Sequential(                      # input (B, 3, 32, 32)
            nn.Conv2d(3, 32, kernel_size=3, padding=1),     # (B, 32, 32, 32)
            nn.ReLU(),
            nn.MaxPool2d(2),                                # (B, 32, 16, 16)
            nn.Conv2d(32, 64, kernel_size=3, padding=1),    # (B, 64, 16, 16)
            nn.ReLU(),
            nn.MaxPool2d(2),                                # (B, 64, 8, 8)
        )
        self.classifier = nn.Sequential(
            nn.Flatten(),                                   # (B, 64*8*8 = 4096)
            nn.Linear(64 * 8 * 8, 128), nn.ReLU(),
            nn.Dropout(0.5),
            nn.Linear(128, num_classes),                    # raw scores (logits)
        )

    def forward(self, x):
        return self.classifier(self.features(x))

Memory hook: kernel 3 + padding 1 keeps the size; each MaxPool2d(2) halves it. Linear input = channels × H × W after the last pool.

CNN-2Write train_one_epoch and evaluate (accuracy) for a multi-class CNN.
Answer
def train_one_epoch(model, loader, loss_fn, optimizer):
    model.train()
    total_loss = 0.0
    for xb, yb in loader:
        optimizer.zero_grad()
        loss = loss_fn(model(xb), yb)      # loss_fn = nn.CrossEntropyLoss()
        loss.backward()
        optimizer.step()
        total_loss += loss.item() * xb.size(0)
    return total_loss / len(loader.dataset)

def evaluate(model, loader):
    model.eval()                            # turns off Dropout
    correct, total = 0, 0
    with torch.no_grad():                   # no gradients needed
        for xb, yb in loader:
            preds = model(xb).argmax(dim=1)
            correct += (preds == yb).sum().item()
            total += yb.size(0)
    return correct / total

model = SimpleCNN()
loss_fn = nn.CrossEntropyLoss()
optimizer = torch.optim.Adam(model.parameters(), lr=1e-3)
for epoch in range(10):
    train_loss = train_one_epoch(model, train_loader, loss_fn, optimizer)

6. RNN & Variants

RNN-1Write a vanilla RNN by hand (nn.Linear only) that reads input (B, T, D) and returns all hidden states and the last one.
Answer
import torch
import torch.nn as nn

class SimpleRNN(nn.Module):
    def __init__(self, input_size, hidden_size):
        super().__init__()
        self.hidden_size = hidden_size
        self.W_x = nn.Linear(input_size, hidden_size)               # includes bias b
        self.W_h = nn.Linear(hidden_size, hidden_size, bias=False)

    def forward(self, x):                        # x: (B, T, D)
        B, T, _ = x.shape
        h = torch.zeros(B, self.hidden_size)     # h_0 = 0
        outputs = []
        for t in range(T):                       # same weights at every step
            h = torch.tanh(self.W_x(x[:, t, :]) + self.W_h(h))
            outputs.append(h)
        return torch.stack(outputs, dim=1), h    # (B, T, H), (B, H)
RNN-2Write an LSTM cell and a GRU cell by hand (the Lab 11 way).
Answer
class LSTMCell(nn.Module):
    def __init__(self, input_size, hidden_size):
        super().__init__()
        n = input_size + hidden_size
        self.f = nn.Linear(n, hidden_size)      # forget gate
        self.i = nn.Linear(n, hidden_size)      # input gate
        self.g = nn.Linear(n, hidden_size)      # candidate
        self.o = nn.Linear(n, hidden_size)      # output gate

    def forward(self, x_t, state):
        h, c = state
        xh = torch.cat([x_t, h], dim=-1)
        f = torch.sigmoid(self.f(xh))
        i = torch.sigmoid(self.i(xh))
        g = torch.tanh(self.g(xh))
        o = torch.sigmoid(self.o(xh))
        c = f * c + i * g                       # additive cell update
        h = o * torch.tanh(c)
        return h, c

class GRUCell(nn.Module):
    def __init__(self, input_size, hidden_size):
        super().__init__()
        self.W_ir = nn.Linear(input_size, hidden_size); self.W_hr = nn.Linear(hidden_size, hidden_size)
        self.W_iz = nn.Linear(input_size, hidden_size); self.W_hz = nn.Linear(hidden_size, hidden_size)
        self.W_in = nn.Linear(input_size, hidden_size); self.W_hn = nn.Linear(hidden_size, hidden_size)

    def forward(self, x_t, h):
        r = torch.sigmoid(self.W_ir(x_t) + self.W_hr(h))      # reset gate
        z = torch.sigmoid(self.W_iz(x_t) + self.W_hz(h))      # update gate
        n = torch.tanh(self.W_in(x_t) + r * self.W_hn(h))     # candidate
        return (1 - z) * n + z * h

Memory hook: gates use sigmoid (0 to 1 valves), candidates use tanh. LSTM returns (h, c); GRU returns only h.

RNN-3Write a next-token model (Embedding → RNN → Linear) and one training step with gradient clipping.
Answer
class NextTokenModel(nn.Module):
    def __init__(self, vocab_size, emb_dim, hidden_size):
        super().__init__()
        self.embedding = nn.Embedding(vocab_size, emb_dim)
        self.rnn = SimpleRNN(emb_dim, hidden_size)
        self.out = nn.Linear(hidden_size, vocab_size)

    def forward(self, tokens):                   # tokens: (B, T) integer ids
        outputs, _ = self.rnn(self.embedding(tokens))   # (B, T, H)
        return self.out(outputs)                 # (B, T, V) logits

model = NextTokenModel(vocab_size=V, emb_dim=32, hidden_size=64)
criterion = nn.CrossEntropyLoss()
optimizer = torch.optim.Adam(model.parameters(), lr=3e-3)

for x, y in train_loader:                        # y = x shifted left by 1
    optimizer.zero_grad()
    logits = model(x)
    loss = criterion(logits.reshape(-1, V), y.reshape(-1))   # flatten B*T
    loss.backward()
    torch.nn.utils.clip_grad_norm_(model.parameters(), max_norm=1.0)  # stop exploding
    optimizer.step()

T-BPTT: when carrying the hidden state into the next chunk, use h = h.detach() so gradients don't flow back forever.

RNN-4If built-in layers are allowed: sequence classifier with nn.LSTM (e.g. sentiment) using the last hidden state.
Answer
class LSTMClassifier(nn.Module):
    def __init__(self, vocab_size, emb_dim, hidden_size, num_classes, bidirectional=False):
        super().__init__()
        self.embedding = nn.Embedding(vocab_size, emb_dim)
        self.lstm = nn.LSTM(emb_dim, hidden_size, batch_first=True, bidirectional=bidirectional)
        out_dim = hidden_size * (2 if bidirectional else 1)
        self.fc = nn.Linear(out_dim, num_classes)

    def forward(self, tokens):                       # (B, T)
        out, (h_n, c_n) = self.lstm(self.embedding(tokens))
        if self.lstm.bidirectional:
            last = torch.cat([h_n[-2], h_n[-1]], dim=1)   # forward + backward
        else:
            last = h_n[-1]                               # (B, H)
        return self.fc(last)                         # (B, num_classes)

batch_first=True means input is (B, T, D). Swap nn.LSTM → nn.GRU / nn.RNN: those return out, h_n (no c_n).
